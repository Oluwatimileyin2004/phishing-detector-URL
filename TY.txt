import re
import ipaddress
import datetime
import whois
import socket
from urllib.parse import urlparse

def analyze_url_structure(url):
    """Analyzes the URL structure and returns a suspicion score."""
    score = 0
    reasons = []
    
    parsed = urlparse(url)
    domain = parsed.netloc
    path = parsed.path
    
    # Check 1: IP address instead of domain (+2)
    try:
        ipaddress.ip_address(domain)
        score += 2
        reasons.append("Uses raw IP address (+2)")
    except ValueError:
        pass  
    
    # Check 2: Excessive subdomains or hyphens (+3)
    dot_count = domain.count('.')
    if dot_count >= 3:
        score += 3
        reasons.append(f" Excessive subdomains ({dot_count} dots) (+3)")
    
    # Check for hyphens in the brand name
    domain_parts = domain.split('.')
    if len(domain_parts) >= 2:
        brand_name = domain_parts[-2] 
        if '-' in brand_name:
            score += 3
            reasons.append("➖ Contains hyphens in brand name (+3)")
    
    # Check 3: Contains "login" or "verify" (+1)
    if "login" in url.lower() or "verify" in url.lower() or "signin" in url.lower():
        score += 1
        reasons.append(" Contains 'login' or 'verify' (+1)")
    
    return score, reasons, domain

def check_domain_age(domain):
    """Checks domain age with a timeout specifically for Kali's network."""
    score = 0
    reason = ""
    
    try:
        # Set a short timeout so Kali doesn't freeze if WHOIS is blocked
        socket.setdefaulttimeout(5) 
        domain_info = whois.whois(domain)
        creation_date = domain_info.creation_date
        
        if isinstance(creation_date, list):
            creation_date = creation_date[0]
        
        if creation_date:
            age_in_days = (datetime.datetime.now() - creation_date).days
            
            if age_in_days < 30:
                score += 4
                reason = f"Domain is only {age_in_days} days old (+4)"
            else:
                reason = f"Domain is {age_in_days} days old (safe)"
        else:
            reason = "❓ No creation date found"
            
    except socket.timeout:
        reason = "WHOIS request timed out (Kali network) - No points added"
    except Exception as e:
        # Catch-all for any other Kali-specific errors
        reason = f"WHOIS skipped (Error: {type(e).__name__}) - No points added"
    
    return score, reason

def check_url(url):
    """The main function to check a URL."""
    print(f"\n🔍 Analyzing: {url}")
    print("-" * 50)
    
    struct_score, reasons, domain = analyze_url_structure(url)
    whois_score, whois_reason = check_domain_age(domain)
    
    total_score = struct_score + whois_score
    
    for reason in reasons:
        print(f"  {reason}")
    print(f"  {whois_reason}")
    
    print("-" * 50)
    print(f"📊 TOTAL SUSPICION SCORE: {total_score} / 10")
    
    if total_score >= 7:
        print("🔴 VERDICT: HIGH RISK - Do NOT click!")
    elif total_score >= 4:
        print("🟡 VERDICT: MEDIUM RISK - Caution!")
    else:
        print("🟢 VERDICT: LOW RISK - Looks safe.")
    
    print("=" * 50)
    return total_score
